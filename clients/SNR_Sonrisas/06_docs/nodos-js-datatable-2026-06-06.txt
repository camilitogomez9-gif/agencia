const fechasInvalidas = [];
for (const item of items) {
  const r = item.json || {};
  const cod = String(r["Código Prestación"] || "").trim();
  const pg = Number(r["Pagado Prestación"]);
  const tieneContenido = cod !== "" || (Number.isFinite(pg) && pg !== 0);
  if (tieneContenido && r._fechaValida === false) {
    const fechaCruda = r["Fecha_de_recepcion_del_pago"] ?? r["Fecha de recepción del pago"];
    const idFila = r.id ?? r.row_number;
    fechasInvalidas.push((idFila != null ? ("fila " + idFila + " ") : "") + "cod=" + cod + " fecha='" + String(fechaCruda ?? "") + "'");
  }
}

if (desconocidos.size > 0 || fechasInvalidas.length > 0) {
  const lista = [...desconocidos].sort().join(", ");
  return [{ json: { _error: true, codigos: lista, fechas_invalidas: fechasInvalidas } }];
}

// Todos válidos → agregar flags
return items.map(item => {
  const row = item.json || {};

      ...row,
      _error: false,
      computaImporte: !noComputaImporte.has(codigo),
      computaCantidad: !noComputaCantidad.has(codigo),
    },
  };
});

    json: {
      ...row,
      Clase: clase,
      cantidadComputable: noSumanCantidad.has(codigo) ? 0 : 1,
    },
  });
